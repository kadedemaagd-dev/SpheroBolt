async def start_program():

    #Purple C-1
    set_main_led({'r': 0, 'g': 0, 'b': 0})
    set_speed(60)
    await delay(2.1)
    set_speed(0)
    set_main_led({'r': 255, 'g': 0, 'b': 255})
    
    #Green C-2
    await delay(2)
    set_main_led({'r': 0, 'g': 0, 'b': 0})
    await roll((get_heading() + 90), 55, 1.5)
    set_main_led({'r': 0, 'g': 255, 'b': 0})

    #Noise C-3
    await delay(2)
    set_main_led({'r': 0, 'g': 0, 'b': 0})
    await roll((get_heading() + 90), 55, 1)
    await delay(1)
    await roll((get_heading() + 45), 45, 1)
    await speak("sounds", True)
    
    #Emoji C-4
    await delay(1)
    await roll((get_heading() - 90), 45, 1)
    await delay(1)
    await roll((get_heading() - 90), 55, 1)
    play_matrix_animation(0, True)
    await delay(1)
    await roll((get_heading() + 90), 30, 1)
    await delay(2)

    #Blue C-5
    await delay(1)
    await roll((get_heading() + 90), 40, 1)
    set_main_led({'r': 0, 'g': 0, 'b': 200})
    
    #Orange C-6
    await delay(1)
    await roll((get_heading() - 135), 50, 1)
    set_main_led({'r': 255, 'g': 166, 'b': 0})
    await delay(1)
    set_main_led({'r': 0, 'g': 0, 'b': 0})

    #Rotate CW C-7
    await roll((get_heading() - 45), 40, 1)
    await delay(1)
    await spin(360, 4)

    #Sound 2 C-8
    await delay(1)
    await roll((get_heading() - 90), 65, 1)
    await speak("sounds part 2", True)

    #Rotate CCW C-9
    await delay(2)
    await roll((get_heading() + 45), 60, 1)
    await spin(-360, 4)

    #BONUS - VICTORY C-10
    await delay(1)
    await set_speed(20)
    await delay(1)
    await set_speed(0)
    set_main_led({'r': 0, 'g': 255, 'b': 0})
